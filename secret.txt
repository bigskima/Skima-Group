Provider activation SQL

Run this in Supabase SQL Editor after secrets are set:

```sql
update public.provider_adapters
set status = 'active', updated_at = timezone('utc', now())
where key in (
  'provider.payment.paystack',
  'provider.maps.google-maps',
  'provider.ai.google-gemini',
  'provider.communication.resend'
);

update public.configuration_entries
set value = jsonb_set(value, '{active_provider_key}', '"provider.maps.google-maps"'::jsonb),
    version = version + 1,
    updated_at = timezone('utc', now())
where namespace = 'platform.maps'
  and key = 'provider_selection'
  and scope_type = 'global'
  and status = 'active';

update public.configuration_entries
set value = jsonb_set(value, '{active_provider_key}', '"provider.ai.google-gemini"'::jsonb),
    version = version + 1,
    updated_at = timezone('utc', now())
where namespace = 'platform.ai'
  and key = 'provider_selection'
  and scope_type = 'global'
  and status = 'active';

update public.configuration_entries
set value = jsonb_set(value, '{active_provider_key}', '"provider.payment.paystack"'::jsonb),
    version = version + 1,
    updated_at = timezone('utc', now())
where namespace = 'platform.payments'
  and key = 'provider_selection'
  and scope_type = 'global'
  and status = 'active';
```
